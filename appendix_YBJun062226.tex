\section{Technical Details}
%---------------------------
\subsection{Common Single-index models}
In this section we discuss the estimation of common single-index models, including partially linear single-index models, partially linear logistic single-index models, and partially linear Poisson single-index models. 

%------linear modle------; 
\emph{Partially Linear \sout{Logistic} Single-Index Models.}
Consider partially linear single-index models
\begin{equation}
	y_i =\mu_i+\ve_i =  \Bb^\p\bx_i+\Bz^\p\bb_i+\ve_i,~~~\ve_i\sim N(0,\sigma^2)\label{linear}
\end{equation}
The penalized log-likelihood of $\Bt$ is given by
\begin{equation*}
	l_\lambda(\Bt) = -\frac{1}{2\sigma^2}\sum_{i=1}^n \big(y_i-\mu_i\big)^2 - \frac{1}{2\sigma^2}\lambda\Bz^\p\BD^\p\BD\Bz.
\end{equation*}	
%--------------------------
In the following are gradient and Fisher of $l_\lambda(\Bt)$ and robust variance estimators of $\wh\Bt$:
\vspace{-0.35cm}
\begin{itemize}
	\item The gradient of $l_\lambda(\Bt)$ is 
	$$
	\nabla l_\lambda(\Bt) = 1/\sigma^2\BE^\p(\by-\Bm)-\lambda/\sigma^2\BP\Bt.
	$$ 
	\item  The Fisher information of $l_\lambda(\Bt)$ is
	$$
	\BI_\lambda(\Bt) = 1/\sigma^2\BE^\p\BE+\lambda/\sigma^2\BP.
	$$
	\item For model \eqref{linear}, $\tg(\mu_i) = \mu_i$, $V(\mu_i) = 1$, and $\phi = \sigma^2$. Thus, $\BG = \BE$ and $\BV$ is an identity matrix. Further, $\sigma^2$ can be updated via an adjusted Pearson statistics methods, namely,
	$$
	\wh\sigma^2 = \frac{1}{\rho} \sum_{i=1}^n (y_i-\wh\mu_i)^2,
	$$
	where $\x\rho$ is an adjusted degrees of freedom defined as $\text{tr}\left\{\big(\BE^\p\BE+\lambda\BP\big)^{-1}\BE^\p\BE\right\}$. 
	\item For robust variance estimation, $U_i(\Bt) = 1/\sigma^2(y_i-\mu)\BE_i$, and hence,
	$$
	\BH(\Bt) = \sum_{i=1}^n{\h \var(U_i(\Bt))}\cancel{U_i(\Bt)^{\otimes 2}} = 1/\sigma^4\BE^\p\BF\BE, 
	$$
	where \sout{$\BF= \text{diag}\bigl\{(y_1-\mu_1)^2,\ldots,(y_n-\mu_n)^2\bigr\}$.} 
	{\h $\BF=\text{diag}\bigl\{\var(y_1-\mu_1),\ldots,\var(y_n-\mu_n)\bigr\}=\sigma^2\BI$.}
	\item The robust variance of $\wh\Bt$ can be written as
	$$
	{\x \BV_{\lambda,r}(\wh\Bt) = \cancel{\BI_\lambda^{-1}(\wh\Bt)\BH(\wh\Bt)\BI_\lambda^{-1}(\wh\Bt)}
		= {\h\BI_\lambda^{-1}(\Bt)\BH(\Bt)\BI_\lambda^{-1}(\Bt)}
		= \bigl(\BE^\p\BE+\lambda\BP\bigr)^{-1}\bigl(\BE^\p\BF\BE\bigr)\bigl(\BE^\p\BE+\lambda\BP\bigr)^{-1}.}
	$$
	\item  For leverage-adjusted robust variance estimation, 
	$$
	\x\BV_{\lambda,c}^{(1)}(\wh\Bt) 
	= \left(\BE\BE^\p+\lambda\BP\right)^{-1}\left\{\sum_{i=1}^n\left(\frac{y_i-\wh\mu_i}{1-h_{ii}}\right)^2\BE_i^{\otimes 2}\right\}\left(\BE\BE^\p+\lambda\BP\right)^{-1},
	$$
	where
	$
	h_{ii} =\BE_i^\p\BA\BE_i
	$
	with 
	$\BA  = \left(\ds\sum_{i=1}^n\BE_i^{\otimes 2}\right)^{-1},
	$
	\item  The sample correction robust variance estimator is given by
	$$
	\x\BV_{\lambda,c}^{(2)}(\wh\Bt) = \BV_{\lambda,r}(\wh\Bt)+\kappa\delta(\BE\BE^\p+\lambda\BP)^{-1},
	$$
	where 
	$
	\delta = \min\left\{0.5,\rho/(n-\rho)\right\}
	$
	and
	\sout{$\kappa = \max\left[1,\text{trace}\left\{(\BE\BE^\p+\lambda\BP)^{-1}\BH(\Bt)\right\}\right]$.}
	{\h
\begin{align*}
\kappa 
&= \max\left[1,\rho^{-1}\text{trace}\left\{\BI_{\lambda}^{-1}(\Bt)\BH(\Bt)\right\}\right] \\
&= \max\left[1,\rho^{-1}\text{trace}\left\{(1/\sigma^2\BE\BE^\p+\lambda/\sigma^2\BP)^{-1}\BH(\Bt)\right\}\right] \\
&= \max\left[1,\rho^{-1}\sigma^2\text{trace}\left\{(\BE\BE^\p+\lambda\BP)^{-1}\BH(\Bt)\right\}\right] 
\end{align*}
}
\end{itemize}

%------logistic modle------; 
\emph{Partially Linear Logistic Single-Index Models.}
Consider partially linear logistic single-index models
\begin{equation}
	\log \frac{\mu_i}{1-\mu_i} = \Bb^\p\bz_i+\Bz^\p\bb_i\Longleftrightarrow \mu_i = \frac{\exp(\Bb^\p\bx_i+\Bz^\p\bb_i)}{1+\exp(\Bb^\p\bx_i+\Bz^\p\bb_i)}. \label{logistic}
\end{equation}
The penalized log-likelihood of $\Bt$ is given by
\begin{equation*}
	l_\lambda(\Bt) = \sum_{i=1}^n \left\{y_i\log\mu_i+\big(1-y_i\big)\log\big(1-\mu_i\big)\right\} - \frac{1}{2}\lambda\Bz^\p\BD^\p\BD\Bz.
\end{equation*}	
%--------------------------
In the following are gradient and Fisher of $l_\lambda(\Bt)$ and robust variance estimators of $\wh\Bt$:
\vspace{-0.35cm}
\begin{itemize}
\item The gradient of $l_\lambda(\Bt)$ is 
$$
\nabla l_\lambda(\Bt) = \BE^\p(\by-\Bm)-\lambda\BP\Bt
$$ 
\item The Fisher information of $l_\lambda(\Bt)$ is
$$
\BI_\lambda(\Bt) = \BE^\p\BV_1\BE+\lambda\BP,
$$
where $\BV_1 = \text{diag}\left\{\mu_1(1-\mu_1),\ldots,\mu_n(1-\mu_n)\right\}$. 
\item For model \eqref{logistic}, $\tg(\mu_i) = \ds\log\frac{\mu_i}{1-\mu_i}$, $V(\mu_i) = \mu_i(1-\mu_i)$, and $\phi = 1$. Thus, $\BG = \BV_1\BE$. 
\item For robust variance estimation, $U_i(\Bt) = (y_i-\mu_i)\BE_i$, and hence	
$$
\BH(\Bt) = \sum_{i=1}^n \cancel{U_i(\Bt)^{\otimes 2}}
{\h\var(U_i(\Bt))}
= \BE^\p\BF\BE,
$$
{\h where $\BF=\text{diag}\left\{\var(y_1-\mu_1),\ldots,\var(y_n-\mu_n)\right\}
	=\text{diag}\left\{\mu_1(1-\mu_1),\ldots,\mu_n(1-\mu_n)\right\}=\BV_1$.}
\item The robust variance of $\wh\Bt$ can be written as
$$
{\x\BV_{\lambda,r}(\wh\Bt) = \cancel{\BI_\lambda^{-1}(\wh\Bt)\BH(\wh\Bt)\BI_\lambda^{-1}(\wh\Bt)}
={\h	\BI_\lambda^{-1}(\Bt)\BH(\Bt)\BI_\lambda^{-1}(\Bt)}
= \bigl(\BE^\p\BV_1\BE+\lambda\BP\bigr)^{-1}\bigl(\BE^\p\BF\BE\bigr)\bigl(\BE^\p\BV_1\BE+\lambda\BP\bigr)^{-1}.}
$$
\item For leverage-adjusted robust variance estimation, 
$$
\x\BV_{\lambda,c}^{(1)}(\wh\Bt) 
= \left(\BE\BV_1\BE^\p+\lambda\BP\right)^{-1}\left\{\sum_{i=1}^n\left(\frac{y_i-\wh\mu_i}{1-h_{ii}}\right)^2\BE_i^{\otimes 2}\right\}\left(\BE\BV_1\BE^\p+\lambda\BP\right)^{-1},
$$
where 
$
h_{ii} =\mu_i(1-\mu_i)\BE_i^\p\BA\BE_i
$
with 
$\BA  = \left(\ds\sum_{i=1}^n\mu_i(1-\mu_i)\BE_i^{\otimes 2}\right)^{-1}.
$
\item The sample correction robust variance estimator is given by
$$
\x\BV_{\lambda,c}^{(2)}(\wh\Bt) = \BV_{\lambda,r}(\wh\Bt)+\delta\kappa(\BE\BV_1\BE^\p+\lambda\BP)^{-1},
$$
where 
$
\delta = \min\left\{0.5,\rho/(n-\rho)\right\}
$
and
\sout{$\kappa = \max\left[1,\text{trace}\left\{(\BE\BV_1\BE^\p+\lambda\BP)^{-1}\BH(\Bt)\right\}\right]$.}
{\h
	\begin{align*}
		\kappa 
		&= \max\left[1,\rho^{-1}\text{trace}\left\{\BI_{\lambda}^{-1}(\Bt)\BH(\Bt)\right\}\right] \\
		&= \max\left[1,\rho^{-1}\text{trace}\left\{(\BE\BV_1\BE^\p+\lambda\BP)^{-1}\BH(\Bt)\right\}\right] 
	\end{align*}
}

\end{itemize}
%--------------------------

%------------Poisson model-----------;
\emph{Partially Linear Poisson Single-Index Models.}
Consider partially linear Poisson single-index models
\begin{equation}
	\log\mu_i = \log C_i+\Bb^\p\bz_i+\Bz^\p\bb_i \Longleftrightarrow \mu_i = C_i\exp(\Bb^\p\bz_i+\Bz^\p\bb_i). \label{poisson}
\end{equation}
The penalized log-likelihood of $\Bt$ is given by
\begin{equation*}
	l_\lambda(\Bt) = \sum_{i=1}^n \big(y_i\log\mu_i-\mu_i\big)- \frac{1}{2}\lambda\Bz^\p\BD^\p\BD\Bz.
\end{equation*}	
In the following are gradient and Fisher of $l_\lambda(\Bt)$ and robust variance estimators of $\wh\Bt$:
\vspace{-0.35cm}
\begin{itemize}
	\item The gradient of $l_\lambda(\Bt)$ is
	$$
	\nabla l_\lambda(\Bt) = \BE^\p(\by-\Bm)-\lambda\BP\Bt
	$$
	\item  The Fisher information of $l_\lambda(\Bt)$ is
	$$
	\BI_\lambda(\Bt) = \BE^\p\BV_2 \BE+\lambda\BP,
	$$
	where $\BV_2 = \text{diag}\left\{\mu_1,\ldots,\mu_n\right\}$. 
	\item For model \eqref{poisson}, $\tg(\mu_i) = \ds\log\mu_i$, $V(\mu_i) = \mu_i$, and $\phi = 1$. Thus, $\BG = \BV_2\BE$. 
	\item For robust variance estimation, $U_i(\Bt) = (y_i-\mu_i)\BE_i$, and hence
	$$
	\BH(\Bt) = \cancel{\sum_{i=1}^n U_i(\Bt)^{\otimes 2}= \sum_{i=1}^n(y_i-\mu_i)^2\BE_i^{\otimes 2}}
	={\h  \sum_{i=1}^n \var(U_i(\Bt))=\sum_{i=1}^n\var(y_i-\mu_i)\BE_i^{\otimes 2}}
	= \BE^\p\BF\BE, 
	$$
{\h
where $\BF=\text{dig}\{\var(y_1-\mu_1),\dots,\var(y_n-\mu_n)\}=\text{dig}\{\mu_1,\dots,\mu_n\}=\BV_2$
}	
	
	\item The robust variance of $\wh\Bt$ can be written as 
	$$
	{\x\BV_{\lambda,r}(\wh\Bt) = \cancel{\BI_\lambda(\wh\Bt)^{-1}\BH(\wh\Bt)\BI_\lambda^{-1}(\wh\Bt)}
		={\h\BI_\lambda(\Bt)^{-1}\BH(\Bt)\BI_\lambda^{-1}(\Bt)}
		= \bigl(\BE^\p\BV_2\BE+\lambda\BP\bigr)^{-1}\bigl(\BE^\p\BF\BE\bigr)\bigl(\BE^\p\BV_2\BE+\lambda\BP\bigr)^{-1}.}
	$$
	\item For leverage-adjusted robust variance estimation, 
	$$
	\x\BV_{\lambda,c}^{(1)}(\wh\Bt) 
	= \left(\BE\BV_2\BE^\p+\lambda\BP\right)^{-1}\left\{\sum_{i=1}^n\left(\frac{y_i-\wh\mu_i}{1-h_{ii}}\right)^2\BE_i^{\otimes 2}\right\}\left(\BE\BV_2\BE^\p+\lambda\BP\right)^{-1},
	$$
	where 
	$
	h_{ii} =\mu_i\BE_i^\p\BA\BE_i
	$
	with 
	$\BA  = \left(\ds\sum_{i=1}^n\mu_i\BE_i^{\otimes 2}\right)^{-1}
	$
	\item The sample correction robust variance estimator is given by
	$$
	\x\BV_{\lambda,c}^{(2)}(\wh\Bt) = \BV_{\lambda,r}(\wh\Bt)+\delta\kappa(\BE\BV_2\BE^\p+\lambda\BP)^{-1},
	$$
	where 
	$\delta = \min\left\{0.5,\rho/(n-\rho)\right\}$ and $\kappa = \max\left[1,{\h\rho^{-1}}\text{trace}\left\{(\BE\BV_2\BE^\p+\lambda\BP)^{-1}\BH(\Bt)\right\}\right].$	
\end{itemize}
